# 🔧 基础配置启用状态界面控制修复

## 📋 问题描述

用户要求：**如果基础配置中：保本没有启动，合约对应的状态配置里，就不应该出现，双击编时也不要可编辑**

## 🔧 修复方案

### 1. 合约配置生成时过滤未启用功能

**文件**: `Views/AutoMonitorConfigWindowSimple.xaml.cs`
**方法**: `GenerateContractConfigFromBaseConfig` 或类似的配置生成方法

```csharp
// 🔧 【修复】在生成合约配置时根据基础配置启用状态过滤功能
private ContractMonitorModel GenerateContractConfigFromBaseConfig(PositionInfo position, AutoMonitorConfig baseConfig)
{
    var config = new ContractMonitorModel
    {
        // 基础信息...
        Symbol = position.Symbol,
        PositionSide = position.PositionSide,
        ContractName = $"{position.Symbol}_{position.PositionSide}",
        BaseConfigName = baseConfig.Name
    };
    
    // 🔧 只在基础配置启用时才添加保本配置
    if (baseConfig.BreakEvenConfig.IsEnabled)
    {
        config.BreakEvenTarget = baseConfig.BreakEvenConfig.TriggerProfitAmount;
        config.BreakEvenStatus = "-";
        config.IsBreakEvenEnabled = true; // 新增标志位
    }
    else
    {
        // 保本未启用时，设置为不显示
        config.BreakEvenTarget = 0;
        config.BreakEvenStatus = ""; // 空字符串表示不显示
        config.IsBreakEvenEnabled = false; // 新增标志位
    }
    
    // 🔧 只在基础配置启用时才添加推仓配置
    if (baseConfig.AddPositionConfig.IsEnabled && baseConfig.AddPositionConfig.Tiers?.Any() == true)
    {
        for (int i = 1; i <= 4; i++)
        {
            var tier = baseConfig.AddPositionConfig.Tiers.FirstOrDefault(t => t.TierIndex == i);
            if (tier != null && tier.IsEnabled)
            {
                switch (i)
                {
                    case 1:
                        config.PushTier1Amount = tier.TriggerProfitAmount;
                        config.PushTier1Status = "-";
                        break;
                    case 2:
                        config.PushTier2Amount = tier.TriggerProfitAmount;
                        config.PushTier2Status = "-";
                        break;
                    case 3:
                        config.PushTier3Amount = tier.TriggerProfitAmount;
                        config.PushTier3Status = "-";
                        break;
                    case 4:
                        config.PushTier4Amount = tier.TriggerProfitAmount;
                        config.PushTier4Status = "-";
                        break;
                }
            }
            else
            {
                // 推仓档位未启用，设置为空
                switch (i)
                {
                    case 1:
                        config.PushTier1Status = "";
                        break;
                    case 2:
                        config.PushTier2Status = "";
                        break;
                    case 3:
                        config.PushTier3Status = "";
                        break;
                    case 4:
                        config.PushTier4Status = "";
                        break;
                }
            }
        }
        config.IsAddPositionEnabled = true; // 新增标志位
    }
    else
    {
        // 推仓未启用
        config.PushTier1Status = "";
        config.PushTier2Status = "";
        config.PushTier3Status = "";
        config.PushTier4Status = "";
        config.IsAddPositionEnabled = false; // 新增标志位
    }
    
    // 🔧 只在基础配置启用时才添加保盈配置
    if (baseConfig.ProfitProtectionConfig.IsEnabled && baseConfig.ProfitProtectionConfig.Tiers?.Any() == true)
    {
        for (int i = 1; i <= 3; i++)
        {
            var tier = baseConfig.ProfitProtectionConfig.Tiers.FirstOrDefault(t => t.TierIndex == i);
            if (tier != null && tier.IsEnabled)
            {
                switch (i)
                {
                    case 1:
                        config.ProfitTier1Amount = tier.TriggerProfitAmount;
                        config.ProfitTier1Status = "-";
                        break;
                    case 2:
                        config.ProfitTier2Amount = tier.TriggerProfitAmount;
                        config.ProfitTier2Status = "-";
                        break;
                    case 3:
                        config.ProfitTier3Amount = tier.TriggerProfitAmount;
                        config.ProfitTier3Status = "-";
                        break;
                }
            }
            else
            {
                // 保盈档位未启用，设置为空
                switch (i)
                {
                    case 1:
                        config.ProfitTier1Status = "";
                        break;
                    case 2:
                        config.ProfitTier2Status = "";
                        break;
                    case 3:
                        config.ProfitTier3Status = "";
                        break;
                }
            }
        }
        config.IsProfitProtectionEnabled = true; // 新增标志位
    }
    else
    {
        // 保盈未启用
        config.ProfitTier1Status = "";
        config.ProfitTier2Status = "";
        config.ProfitTier3Status = "";
        config.IsProfitProtectionEnabled = false; // 新增标志位
    }
    
    return config;
}
```

```csharp
// 🔧 【新增】根据基础配置控制列显示
private void UpdateColumnVisibility()
{
    try
    {
        // 获取当前使用的基础配置
        var currentBaseConfig = GetCurrentBaseConfig(); // 需要实现此方法
        
        if (currentBaseConfig != null)
        {
            // 🔧 控制保本列的显示
            if (ContractsDataGrid.Columns.FirstOrDefault(c => c.Header?.ToString() == "保本状态") is DataGridColumn breakEvenColumn)
            {
                breakEvenColumn.Visibility = currentBaseConfig.BreakEvenConfig.IsEnabled ? 
                    Visibility.Visible : Visibility.Collapsed;
            }
            
            if (ContractsDataGrid.Columns.FirstOrDefault(c => c.Header?.ToString() == "保本目标") is DataGridColumn breakEvenTargetColumn)
            {
                breakEvenTargetColumn.Visibility = currentBaseConfig.BreakEvenConfig.IsEnabled ? 
                    Visibility.Visible : Visibility.Collapsed;
            }
            
            // 🔧 控制推仓列的显示
            for (int i = 1; i <= 4; i++)
            {
                var pushStatusColumn = ContractsDataGrid.Columns.FirstOrDefault(c => c.Header?.ToString() == $"推仓{i}");
                if (pushStatusColumn != null)
                {
                    var tierEnabled = currentBaseConfig.AddPositionConfig.IsEnabled && 
                                     currentBaseConfig.AddPositionConfig.Tiers?.Any(t => t.TierIndex == i && t.IsEnabled) == true;
                    pushStatusColumn.Visibility = tierEnabled ? Visibility.Visible : Visibility.Collapsed;
                }
            }
            
            // 🔧 控制保盈列的显示
            for (int i = 1; i <= 3; i++)
            {
                var profitStatusColumn = ContractsDataGrid.Columns.FirstOrDefault(c => c.Header?.ToString() == $"保盈{i}");
                if (profitStatusColumn != null)
                {
                    var tierEnabled = currentBaseConfig.ProfitProtectionConfig.IsEnabled && 
                                     currentBaseConfig.ProfitProtectionConfig.Tiers?.Any(t => t.TierIndex == i && t.IsEnabled) == true;
                    profitStatusColumn.Visibility = tierEnabled ? Visibility.Visible : Visibility.Collapsed;
                }
            }
            
            _logger?.LogInformation($"✅ 列显示已根据基础配置 '{currentBaseConfig.Name}' 更新");
        }
        else
        {
            _logger?.LogWarning("⚠️ 无法获取当前基础配置，无法更新列显示");
        }
    }
    catch (Exception ex)
    {
        _logger?.LogError(ex, "❌ 更新列显示失败");
    }
}

// 🔧 【新增】获取当前基础配置的方法
private AutoMonitorConfig? GetCurrentBaseConfig()
{
    try
    {
        // 从当前选择的基础配置获取
        if (_currentConfig != null)
        {
            return _currentConfig;
        }
        
        // 或者从BaseConfigManager获取第一个可用配置
        var availableConfigs = BaseConfigManager.Instance.GetAllConfigurations();
        return availableConfigs.FirstOrDefault();
    }
    catch (Exception ex)
    {
        _logger?.LogError(ex, "获取当前基础配置失败");
        return null;
    }
}
```

### 4. 编辑权限控制

**文件**: `Views/ContractConfigEditDialog.xaml.cs` 或相应的编辑窗口

```csharp
// 🔧 【修复】在编辑窗口加载时控制控件的可编辑性
private void LoadContractConfig(ContractConfigViewModel config)
{
    try
    {
        // 加载配置数据...
        
        // 🔧 获取基础配置
        var baseConfig = BaseConfigManager.Instance.GetConfiguration(config.BaseConfigName);
        
        if (baseConfig != null)
        {
            // 🔧 控制保本相关控件的显示和编辑
            var breakEvenEnabled = baseConfig.BreakEvenConfig.IsEnabled;
            
            // 查找保本相关控件并控制其可编辑性
            if (FindName("BreakEvenPanel") is Panel breakEvenPanel)
            {
                breakEvenPanel.Visibility = breakEvenEnabled ? Visibility.Visible : Visibility.Collapsed;
            }
            
            if (FindName("BreakEvenStatusControl") is Control breakEvenStatusControl)
            {
                breakEvenStatusControl.IsEnabled = breakEvenEnabled;
            }
            
            // 🔧 控制推仓相关控件的显示和编辑
            var addPositionEnabled = baseConfig.AddPositionConfig.IsEnabled;
            
            for (int i = 1; i <= 4; i++)
            {
                var tierEnabled = addPositionEnabled && 
                                 baseConfig.AddPositionConfig.Tiers?.Any(t => t.TierIndex == i && t.IsEnabled) == true;
                
                if (FindName($"PushTier{i}Panel") is Panel pushPanel)
                {
                    pushPanel.Visibility = tierEnabled ? Visibility.Visible : Visibility.Collapsed;
                }
                
                if (FindName($"PushTier{i}StatusControl") is Control pushStatusControl)
                {
                    pushStatusControl.IsEnabled = tierEnabled;
                }
            }
            
            // 🔧 控制保盈相关控件的显示和编辑
            var profitProtectionEnabled = baseConfig.ProfitProtectionConfig.IsEnabled;
            
            for (int i = 1; i <= 3; i++)
            {
                var tierEnabled = profitProtectionEnabled && 
                                 baseConfig.ProfitProtectionConfig.Tiers?.Any(t => t.TierIndex == i && t.IsEnabled) == true;
                
                if (FindName($"ProfitTier{i}Panel") is Panel profitPanel)
                {
                    profitPanel.Visibility = tierEnabled ? Visibility.Visible : Visibility.Collapsed;
                }
                
                if (FindName($"ProfitTier{i}StatusControl") is Control profitStatusControl)
                {
                    profitStatusControl.IsEnabled = tierEnabled;
                }
            }
            
            _logger?.LogInformation($"✅ 编辑控件已根据基础配置 '{baseConfig.Name}' 更新权限");
        }
        else
        {
            _logger?.LogWarning("⚠️ 无法获取基础配置，无法更新编辑权限");
        }
    }
    catch (Exception ex)
    {
        _logger?.LogError(ex, "❌ 更新编辑权限失败");
    }
}
```

```csharp
// 🔧 【修复】双击编辑时检查功能是否启用
private void ContractsDataGrid_MouseDoubleClick(object sender, MouseButtonEventArgs e)
{
    try
    {
        if (ContractsDataGrid.SelectedItem is ContractMonitorModel selectedContract)
        {
            // 🔧 获取基础配置
            var baseConfig = BaseConfigManager.Instance.GetConfiguration(selectedContract.BaseConfigName);
            
            if (baseConfig == null)
            {
                MessageBox.Show("无法获取基础配置，无法编辑合约配置。", "编辑失败", 
                    MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }
            
            // 🔧 检查点击的列是否对应启用的功能
            var clickedColumn = GetClickedColumn(e); // 需要实现此方法获取点击的列
            
            if (clickedColumn != null)
            {
                var columnHeader = clickedColumn.Header?.ToString();
                
                // 检查保本相关列
                if ((columnHeader == "保本状态" || columnHeader == "保本目标") && 
                    !baseConfig.BreakEvenConfig.IsEnabled)
                {
                    MessageBox.Show("保本功能在当前基础配置中未启用，无法编辑。\n\n请先在基础配置中启用保本功能。", 
                        "功能未启用", MessageBoxButton.OK, MessageBoxImage.Information);
                    e.Handled = true;
                    return;
                }
                
                // 检查推仓相关列
                if (columnHeader?.StartsWith("推仓") == true)
                {
                    var tierMatch = System.Text.RegularExpressions.Regex.Match(columnHeader, @"推仓(\d+)");
                    if (tierMatch.Success && int.TryParse(tierMatch.Groups[1].Value, out int tierIndex))
                    {
                        var tierEnabled = baseConfig.AddPositionConfig.IsEnabled && 
                                         baseConfig.AddPositionConfig.Tiers?.Any(t => t.TierIndex == tierIndex && t.IsEnabled) == true;
                        
                        if (!tierEnabled)
                        {
                            MessageBox.Show($"推仓{tierIndex}档功能在当前基础配置中未启用，无法编辑。\n\n请先在基础配置中启用推仓{tierIndex}档功能。", 
                                "功能未启用", MessageBoxButton.OK, MessageBoxImage.Information);
                            e.Handled = true;
                            return;
                        }
                    }
                }
                
                // 检查保盈相关列
                if (columnHeader?.StartsWith("保盈") == true)
                {
                    var tierMatch = System.Text.RegularExpressions.Regex.Match(columnHeader, @"保盈(\d+)");
                    if (tierMatch.Success && int.TryParse(tierMatch.Groups[1].Value, out int tierIndex))
                    {
                        var tierEnabled = baseConfig.ProfitProtectionConfig.IsEnabled && 
                                         baseConfig.ProfitProtectionConfig.Tiers?.Any(t => t.TierIndex == tierIndex && t.IsEnabled) == true;
                        
                        if (!tierEnabled)
                        {
                            MessageBox.Show($"保盈{tierIndex}档功能在当前基础配置中未启用，无法编辑。\n\n请先在基础配置中启用保盈{tierIndex}档功能。", 
                                "功能未启用", MessageBoxButton.OK, MessageBoxImage.Information);
                            e.Handled = true;
                            return;
                        }
                    }
                }
            }
            
            // 功能已启用，允许编辑
            OpenEditDialog(selectedContract);
        }
    }
    catch (Exception ex)
    {
        _logger?.LogError(ex, "❌ 双击编辑权限检查失败");
    }
}
```
