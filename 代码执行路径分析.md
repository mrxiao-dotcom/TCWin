```
│   ├── profile.BreakEvenState.IsTriggered = true
```

### 关键代码片段
```csharp
// AutoMonitorExecutionEngine.cs - ProcessBreakEvenLogicAsync()
var isExecutedInState = _stateManager?.IsOperationExecuted(profile.Symbol, positionSide, "BreakEven") ?? false;
var isExecutedInProfile = profile.BreakEvenState.IsTriggered && 
                         (profile.BreakEvenState.ExecutionStatus == "已执行" || profile.BreakEvenState.ExecutionStatus == "模拟执行");

if (config.IsExecuted || isExecutedInState || isExecutedInProfile)
{
    return null; // 跳过已执行的保本
}

// 执行保本止损
var result = await _tradingService.ExecuteBreakEvenStopLossAsync(profile, config);

if (result?.IsSuccess == true)
{
    // 更新状态
    profile.BreakEvenState.IsTriggered = true;
    config.IsExecuted = true;
    _stateManager?.SaveToPersistence();
    await _profileService.UpdateProfileAsync(profile);
}
```

### 加仓市值计算逻辑
```csharp
// TradingExecutionService.cs - CalculateAddPositionValue()
private decimal CalculateAddPositionValue(ContractProfile profile, ContractAddPositionTier tier, decimal currentPrice)
{
    // 获取账户信息
    var accountEquity = GetAccountEquity();
    var riskTimes = GetRiskTimes();
    var singleRiskCapital = accountEquity / riskTimes;
    
    // 应用阶梯配置
    var riskMultiplier = tier.RiskMultiplier;      // 风险金倍数
    var stopLossRatio = tier.StopLossRatio;        // 止损比例
    
    // 核心公式
    var totalValue = (riskMultiplier * singleRiskCapital) / stopLossRatio;
    
    return totalValue;
}
```

```csharp
public bool IsOperationExecuted(string symbol, string positionSide, string operationType, int? tierIndex = null)
{
    var profile = GetPositionProfile(symbol, positionSide);
    if (profile == null) return false;

    var triggerKey = BuildTriggerKey(symbol, positionSide, operationType, tierIndex);
    return profile.TriggerRecords.ContainsKey(triggerKey);
```

### 键值构建逻辑
```csharp
private string BuildTriggerKey(string symbol, string positionSide, string operationType, int? tierIndex)
{
    if (tierIndex.HasValue)
        return $"{symbol}_{positionSide}_{operationType}_{tierIndex}";
    else
        return $"{symbol}_{positionSide}_{operationType}";
}

// 示例键值:
// BTCUSDT_LONG_BreakEven
// BTCUSDT_LONG_AddPosition_1  
// BTCUSDT_LONG_AddPosition_2
// BTCUSDT_LONG_ProfitProtection_1
```

## 🔄 模拟/实盘模式切换

### 模拟模式判断
```csharp
// TradingExecutionService.cs
private bool IsSimulationMode()
{
    try
    {
        if (_binanceService == null) return true;
        
        // 通过反射检查API配置
        var currentAccount = _binanceService.GetType().GetProperty("CurrentAccount")?.GetValue(_binanceService);
        if (currentAccount != null)
        {
            var apiKey = currentAccount.GetType().GetProperty("ApiKey")?.GetValue(currentAccount) as string;
            var secretKey = currentAccount.GetType().GetProperty("SecretKey")?.GetValue(currentAccount) as string;
            
            // 判断条件
            bool isSimulation = string.IsNullOrEmpty(apiKey) || 
                               string.IsNullOrEmpty(secretKey) ||
                               apiKey.Length < 10 || 
                               secretKey.Length < 10;
            
            return isSimulation;
        }
    }
    catch (Exception)
    {
        return true; // 异常时默认模拟模式
    }
    
    return true;
}
```

### 执行模式分支

## 🎯 关键性能优化点

1. **状态检查缓存**：避免重复的文件读取
2. **批量状态更新**：减少文件I/O次数  
3. **异步执行**：不阻塞定时器线程
4. **内存管理**：及时清理无用状态
5. **错误恢复**：API失败时的优雅降级

这个工作流程确保了自动盯盘系统的稳定运行，同时提供了完整的状态管理和错误恢复机制。
