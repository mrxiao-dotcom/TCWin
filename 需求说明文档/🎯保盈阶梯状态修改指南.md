### 2. **Views/ContractConfigEditDialog.xaml.cs**
在SaveContractConfigToFile方法中，添加保盈阶梯状态的保存逻辑：

```csharp
// 🔧 添加保盈阶梯状态保存
var profitProtection1Status = ((ComboBoxItem)ProfitProtection1StatusComboBox.SelectedItem)?.Tag?.ToString() ?? "-";
var profitProtection2Status = ((ComboBoxItem)ProfitProtection2StatusComboBox.SelectedItem)?.Tag?.ToString() ?? "-";
var profitProtection3Status = ((ComboBoxItem)ProfitProtection3StatusComboBox.SelectedItem)?.Tag?.ToString() ?? "-";

_logger.LogCritical($"🔥 保盈阶梯状态: T1={profitProtection1Status}, T2={profitProtection2Status}, T3={profitProtection3Status}");

// 更新保盈阶梯状态到状态服务
try
{
    // 保盈阶梯1
    if (profitProtection1Status == "√")
    {
        stateService.UpdateExecutionStatus(contractKey, "ProfitProtection", 1, true, 0, "手动设置为executed");
    }
    else if (profitProtection1Status == "-")
    {
        stateService.UpdateExecutionStatus(contractKey, "ProfitProtection", 1, false, 0, "手动重置为waiting");
    }

    // 保盈阶梯2
    if (profitProtection2Status == "√")
    {
        stateService.UpdateExecutionStatus(contractKey, "ProfitProtection", 2, true, 0, "手动设置为executed");
    }
    else if (profitProtection2Status == "-")
    {
        stateService.UpdateExecutionStatus(contractKey, "ProfitProtection", 2, false, 0, "手动重置为waiting");
    }

    // 保盈阶梯3
    if (profitProtection3Status == "√")
    {
        stateService.UpdateExecutionStatus(contractKey, "ProfitProtection", 3, true, 0, "手动设置为executed");
    }
    else if (profitProtection3Status == "-")
    {
        stateService.UpdateExecutionStatus(contractKey, "ProfitProtection", 3, false, 0, "手动重置为waiting");
    }
}
catch (Exception ex)
{
    _logger.LogCritical($"🔥 保盈阶梯状态更新失败: {ex.Message}");
}
```

if (state.ProfitProtectionConfig?.Tiers != null)

    var tier2 = state.ProfitProtectionConfig.Tiers.FirstOrDefault(t => t.TierIndex == 2);
    var tier3 = state.ProfitProtectionConfig.Tiers.FirstOrDefault(t => t.TierIndex == 3);

    config.ProfitProtection1Status = (tier1?.ExecutionState == ExecutionState.Executed) ? "√" : "-";
    config.ProfitProtection2Status = (tier2?.ExecutionState == ExecutionState.Executed) ? "√" : "-";

}
